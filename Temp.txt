LC-1137
class Solution {
    public int tribonacci(int n) {
        int[] f = new int[3+n];
        f[0] = 0;
        f[1] = 1;
        f[2] = 1;
        for (int i = 0; i < n; i++)
            f[i + 3] = f[i] + f[i + 1] + f[i + 2];
        return f[n];
    }
}

LC-2161
class Solution {
    public int[] pivotArray(int[] nums, int pivot) {
        int len = nums.length;
        int[] left = new int[len - 1];
        int[] right = new int[len - 1];
        int l = 0, r = 0;
        int same = 0;
        for (int i = 0; i < len; i++) {
            if (nums[i] > pivot)
                right[r++] = nums[i];
            else if (nums[i] < pivot)
                left[l++] = nums[i];
            else
                same++;
        }
        int k = 0;
        //left
        for (int i = 0; i < l; i++)
            nums[k++] = left[i];
        // same
        for (int i = 0; i < same; i++)
            nums[k++] = pivot;
        for (int i = 0; i < r; i++)
            nums[k++] = right[i];
        return nums;
    }
}

class Solution {
    public int[] pivotArray(int[] nums, int pivot) {

        int n = nums.length;
        int[] result = new int[n];

        int k = 0;
        int equal = 0;

        // Smaller elements
        for (int num : nums) {
            if (num < pivot) {
                result[k++] = num;
            } else if (num == pivot) {
                equal++;
            }
        }

        // Equal elements
        while (equal-- > 0) {
            result[k++] = pivot;
        }

        // Greater elements
        for (int num : nums) {
            if (num > pivot) {
                result[k++] = num;
            }
        }

        return result;
    }
}


LC-1047. Remove All Adjacent Duplicates In String
class Solution {
    public String removeDuplicates(String s) {
        Stack<Character> st = new Stack<>();
        StringBuilder res = new StringBuilder();
        int len = s.length();
        for (int i = 0; i < len; i++) {
            char curr = s.charAt(i);
            if (!st.isEmpty() && st.peek() == curr) {
                st.pop();
            } else {
                st.push(curr);
            }
        }
        while (!st.isEmpty()) {
            res.insert(0, st.peek());
            st.pop();
        }
        return res.toString();
    }
}

class Solution {
    public String removeDuplicates(String s) {

        StringBuilder st = new StringBuilder();

        for (int i = 0; i < s.length(); i++) {

            char curr = s.charAt(i);

            int len = st.length();

            if (len > 0 && st.charAt(len - 1) == curr) {
                st.deleteCharAt(len - 1);
            } else {
                st.append(curr);
            }
        }

        return st.toString();
    }
}

LC-1323. Maximum 69 Number
class Solution {
    public int maximum69Number(int num) {
        Stack<Integer> st = new Stack<>();
        while (num > 0) {
            st.push(num % 10);
            num /= 10;
        }
        boolean is9 = false;
        int ans = 0;
        while (!st.isEmpty()) {
            if (!is9 && st.peek() != 9) {
                is9 = true;
                ans = ans * 10 + 9;
            } else {
                ans = ans * 10 + st.peek();
            }
            st.pop();
        }
        return ans;
    }
}

class Solution {
    public int maximum69Number(int num) {

        char[] digits = String.valueOf(num).toCharArray();

        for (int i = 0; i < digits.length; i++) {

            if (digits[i] == '6') {
                digits[i] = '9';
                break;
            }
        }

        return Integer.parseInt(new String(digits));
    }
}
